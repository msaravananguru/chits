-- =====================================================================
-- DAY 0  -  Chit project: database setup (database: sudardev)
-- Run order:  PART 1 (tables)  ->  PART 2 (plan data)  ->  PART 3 (checks)
--
-- WHAT WE ARE DOING
--   Replace the single old table `sudar_chits` with a structure that can
--   generate the monthly collection sheet (Excel sheet) automatically.
--
--   chit_plans            Rate card per plan (e.g. 1 lakh: pay 5000, then 6000 after taking)
--   chit_plan_receivable  Amount received if the chit is taken in month N (கிடைக்கும் தொகை)
--   chits                 One row per member's chit (plan, joined_date, taken_month)
--   monthly_sheet         One row per month: DRAFT (editable) -> LOCKED (read-only)
--   chit_collections      One row per chit per month: due, paid (வரவு), paid_date,
--                         plus calculated values saved when the month is LOCKED
--
-- KEY RULES
--   * Joined in October = chit month 1; November = month 2, and so on.
--   * taken_month NULL  -> member pays monthly_before_taken every month.
--   * taken in month 3  -> pays monthly_before_taken up to month 3 and
--                          monthly_after_taken from month 4 onward.
--   * A month stays DRAFT while checking; after LOCK it cannot be edited.
--   * Old table `sudar_chits` is NOT touched by this script.
--
-- WARNING: PART 1 drops and recreates the tables. Run only on empty tables.
-- =====================================================================

-- ======================= PART 1 : TABLES =============================
USE sudardev;

-- Drop old tables (child first)
DROP TABLE IF EXISTS chit_collections;
DROP TABLE IF EXISTS monthly_sheet;
DROP TABLE IF EXISTS chits;
DROP TABLE IF EXISTS chit_plan_receivable;
DROP TABLE IF EXISTS chit_plans;

-- 1. Plan rate card (one row per plan: 25K, 50K, 100K, 150K, 200K)
CREATE TABLE chit_plans (
    plan_amount           DECIMAL(12,2) NOT NULL,   -- e.g. 100000
    total_months          INT           NOT NULL,   -- e.g. 21
    monthly_before_taken  DECIMAL(12,2) NOT NULL,   -- e.g. 5000
    monthly_after_taken   DECIMAL(12,2) NOT NULL,   -- e.g. 6000
    PRIMARY KEY (plan_amount)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 2. Amount receivable if the chit is taken in a given month (கிடைக்கும் தொகை)
CREATE TABLE chit_plan_receivable (
    plan_amount        DECIMAL(12,2) NOT NULL,
    month_no           INT           NOT NULL,
    receivable_amount  DECIMAL(12,2) NOT NULL,
    PRIMARY KEY (plan_amount, month_no),
    CONSTRAINT fk_recv_plan FOREIGN KEY (plan_amount) REFERENCES chit_plans (plan_amount)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 3. Chit master (one row per member's chit)
CREATE TABLE chits (
    chit_no       INT           NOT NULL,
    member_name   VARCHAR(200)  NOT NULL,
    plan_amount   DECIMAL(12,2) NOT NULL,
    joined_date   DATE          NOT NULL,            -- date the member joined
    taken_month   INT           NULL,                -- NULL = not taken, 3 = taken in month 3
    status        VARCHAR(10)   NOT NULL DEFAULT 'ACTIVE',   -- ACTIVE / CLOSED
    created_at    TIMESTAMP     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (chit_no),
    CONSTRAINT fk_chits_plan   FOREIGN KEY (plan_amount) REFERENCES chit_plans (plan_amount),
    CONSTRAINT chk_chits_status CHECK (status IN ('ACTIVE','CLOSED')),
    CONSTRAINT chk_chits_taken  CHECK (taken_month IS NULL OR taken_month >= 1)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 4. Monthly sheet (DRAFT / LOCKED)
CREATE TABLE monthly_sheet (
    id           INT          NOT NULL AUTO_INCREMENT,
    sheet_year   INT          NOT NULL,
    sheet_month  INT          NOT NULL,
    status       VARCHAR(10)  NOT NULL DEFAULT 'DRAFT',
    locked_at    TIMESTAMP    NULL,
    created_at   TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uk_sheet_year_month (sheet_year, sheet_month),
    CONSTRAINT chk_sheet_month  CHECK (sheet_month BETWEEN 1 AND 12),
    CONSTRAINT chk_sheet_status CHECK (status IN ('DRAFT','LOCKED'))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- 5. Monthly collections (one row per chit per month)
CREATE TABLE chit_collections (
    id                INT            NOT NULL AUTO_INCREMENT,
    sheet_id          INT            NOT NULL,
    chit_no           INT            NOT NULL,
    paid_month        INT            NOT NULL,                -- current month number of the chit
    due_amount        DECIMAL(12,2)  NOT NULL,                -- கட்ட வேண்டிய தொகை
    paid_amount       DECIMAL(12,2)  NOT NULL DEFAULT 0,      -- வரவு
    paid_date         DATE           NULL,                    -- date the payment was received
    -- snapshot values, filled only when the month is LOCKED
    total_paid        DECIMAL(12,2)  NULL,
    total_due_amount  DECIMAL(12,2)  NULL,
    months_pending    DECIMAL(8,2)   NULL,
    remaining_amount  DECIMAL(12,2)  NULL,
    PRIMARY KEY (id),
    UNIQUE KEY uk_sheet_chit (sheet_id, chit_no),
    CONSTRAINT fk_coll_sheet FOREIGN KEY (sheet_id) REFERENCES monthly_sheet (id),
    CONSTRAINT fk_coll_chit  FOREIGN KEY (chit_no)  REFERENCES chits (chit_no),
    CONSTRAINT chk_coll_paid CHECK (paid_amount >= 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ======================= PART 5 : USERS, CREATED_BY, AUDIT ===========
-- Change 2: users table (password stored as plain text for now; hash it after the project is complete)
USE sudardev;

CREATE TABLE users (
    id          INT           NOT NULL AUTO_INCREMENT,
    username    VARCHAR(100)  NOT NULL,
    password    VARCHAR(255)  NOT NULL,                  -- plain text for now, hash later
    full_name   VARCHAR(200)  NULL,
    role        VARCHAR(10)   NOT NULL DEFAULT 'USER',   -- ADMIN / USER
    status      VARCHAR(10)   NOT NULL DEFAULT 'ACTIVE', -- ACTIVE / INACTIVE (users are never deleted)
    created_at  TIMESTAMP     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uk_users_username (username),
    CONSTRAINT chk_users_role   CHECK (role IN ('ADMIN','USER')),
    CONSTRAINT chk_users_status CHECK (status IN ('ACTIVE','INACTIVE'))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Change 4: audit log (insert only; never updated or deleted)
CREATE TABLE audit_log (
    id            BIGINT        NOT NULL AUTO_INCREMENT,
    event_time    TIMESTAMP(3)  NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
    user_id       INT           NULL,            -- no foreign key on purpose: audit rows must always be savable
    username      VARCHAR(100)  NULL,            -- name at that time
    action        VARCHAR(30)   NOT NULL,        -- CREATE, UPDATE, LOCK, GENERATE, LOGIN, LOGIN_FAILED ...
    entity_name   VARCHAR(50)   NULL,            -- chits, chit_collections, monthly_sheet ...
    entity_id     VARCHAR(50)   NULL,            -- chit_no, collection id ...
    old_values    JSON          NULL,            -- before the change
    new_values    JSON          NULL,            -- after the change
    description   VARCHAR(500)  NULL,
    http_method   VARCHAR(10)   NULL,
    endpoint      VARCHAR(200)  NULL,
    ip_address    VARCHAR(45)   NULL,
    result        VARCHAR(10)   NOT NULL DEFAULT 'SUCCESS',  -- SUCCESS / FAILED
    error_message VARCHAR(500)  NULL,
    PRIMARY KEY (id),
    KEY idx_audit_time   (event_time),
    KEY idx_audit_user   (user_id, event_time),
    KEY idx_audit_entity (entity_name, entity_id),
    KEY idx_audit_action (action)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
