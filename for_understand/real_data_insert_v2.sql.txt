SELECT COUNT(*) AS chits, SUM(opening_paid) AS opening_paid FROM chits;                       -- expect 42 and 0.00
SELECT pay_mode, COUNT(*) AS members FROM chits GROUP BY pay_mode;                            -- expect DAILY 18, SINGLE 24
SELECT chit_no, taken_month FROM chits WHERE taken_month IS NOT NULL ORDER BY chit_no;        -- expect 2(1) 4(2) 14(1) 17(3) 18(3) 34(1) 35(1)
SELECT sheet_year, sheet_month, status FROM monthly_sheet ORDER BY sheet_year, sheet_month;   -- expect Jun, Jul, Aug, Sep all LOCKED
SELECT s.sheet_month, COUNT(*) AS rows_, SUM(c.due_amount) AS due, SUM(c.paid_amount) AS received, SUM(c.total_paid) AS total_paid
FROM chit_collections c JOIN monthly_sheet s ON s.id = c.sheet_id GROUP BY s.sheet_month ORDER BY s.sheet_month;
--   expect  6: 14 rows | 60500 | 36100 | 36100
--           7: 29 rows | 144000 | 139600 | 175700
--           8: 35 rows | 181000 | 157900 | 333600
--           9: 42 rows | 209250 | 183250 | 516850

SELECT DATE_FORMAT(paid_date, '%Y-%m') AS month_, COUNT(*) AS receipts, SUM(amount) AS received
FROM chit_payments GROUP BY DATE_FORMAT(paid_date, '%Y-%m') ORDER BY 1;
--   expect  2026-06: 66 | 36100   2026-07: 187 | 139600   2026-08: 205 | 157900   2026-09: 293 | 183250

-- every row of every final month must equal the sum of its receipts (expect NO rows)
SELECT s.sheet_month, c.chit_no, c.paid_amount, IFNULL(p.total, 0) AS receipts
FROM chit_collections c
JOIN monthly_sheet s ON s.id = c.sheet_id
LEFT JOIN (SELECT chit_no, YEAR(paid_date) y, MONTH(paid_date) m, SUM(amount) total FROM chit_payments GROUP BY chit_no, YEAR(paid_date), MONTH(paid_date)) p
       ON p.chit_no = c.chit_no AND p.y = s.sheet_year AND p.m = s.sheet_month
WHERE c.paid_amount <> IFNULL(p.total, 0);

-- total paid of every chit = sum of its receipts (opening_paid is 0)  (expect NO rows)
SELECT c.chit_no, c.opening_paid + IFNULL(SUM(p.amount), 0) AS from_receipts, l.total_paid AS sep_sheet
FROM chits c
LEFT JOIN chit_payments p ON p.chit_no = c.chit_no
JOIN chit_collections l ON l.chit_no = c.chit_no AND l.sheet_id = @s9
GROUP BY c.chit_no, c.opening_paid, l.total_paid
HAVING from_receipts <> sep_sheet;
